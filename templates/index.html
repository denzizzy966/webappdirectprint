📖 Panduan Penggunaan & Contoh Kode

1. Penggunaan di Web Biasa (HTML / Vanilla JS / React / Vue)

Cukup panggil tag <script src="http://127.0.0.1:18212/static/js/hardware-bridge.js"></script> di HTML Anda: 

<!-- 1. Muat SDK dari Bridge Server Lokal -->
<script src="http://127.0.0.1:18212/static/js/hardware-bridge.js"></script>

<script>
async function initBridge() {
  const bridge = new HardwareBridge({ port: 18212 });
  await bridge.connect();

  // A. Ambil daftar printer yang terdeteksi
  const printerList = await bridge.getPrinters();
  console.log("Printer terdeteksi:", printerList.printers);

  // B. Silent Print Struk Thermal (ESC/POS)
  const rawReceipt = "\x1b\x40\x1b\x61\x01TOKO KITA\n----------------\n1x Kopi   Rp 25.000\n----------------\n\n\x1dV\x42\x00";
  await bridge.printRaw("EPSON TM-T82", rawReceipt);

  // C. Silent Print PDF (Faktur / Invoice)
  await bridge.printPdf("Printer Kantor", "http://127.0.0.1:8088/api/invoice/INV-001.pdf");

  // D. Baca Bobot Timbangan Digital (Real-Time)
  const result = await bridge.getWeight();
  console.log("Berat saat ini:", result.weight, result.unit, "Stabil:", result.stable);

  // E. Buka Laci Kasir (Cash Drawer Kick)
  await bridge.openCashDrawer();

  // F. Dengarkan stream perubahan berat secara reactive
  bridge.onWeightChange((scales) => {
    const scale = scales[0];
    document.getElementById("tampilan-berat").innerText = `${scale.weight} ${scale.unit}`;
  });
}
initBridge();
</script>

2. Cara Pasang di ERPNext (Client Script)

  1. Buka Desk ERPNext Anda (sebagai Administrator / System Manager).
  2. Di Search Bar atas, ketik Client Script lalu tekan Enter.
  3. Klik + Add Client Script.
  4. Set DocType ke POS Invoice atau Sales Invoice.
  5. Set Apply To ke Form dan centang Enabled.
  6. Salin kode berikut ke kolom Script dan klik Save:
frappe.ui.form.on('POS Invoice', {
    refresh: function(frm) {
        if (!frm.is_new()) {
            // Tombol Direct Print Struk
            frm.add_custom_button(__('🖨️ Cetak Struk (Direct Print)'), function() {
                frappe.show_alert({ message: __('Mengirim ke printer kasir...'), indicator: 'blue' });

                // Format ESC/POS
                let receipt = "\x1b\x40\x1b\x61\x01\x1d\x21\x11" + (frm.doc.company || 'STRUK KASIR') + "\n\x1d\x21\x00";
                receipt += "----------------------------------------\n\x1b\x61\x00";
                receipt += "No Inv : " + frm.doc.name + "\n";
                receipt += "Waktu  : " + frm.doc.posting_date + "\n========================================\n";

                (frm.doc.items || []).forEach(item => {
                    receipt += `${item.qty}x ${item.item_name} - ${format_currency(item.amount, frm.doc.currency)}\n`;
                });

                receipt += "----------------------------------------\n\x1b\x61\x02";
                receipt += "TOTAL: " + format_currency(frm.doc.grand_total, frm.doc.currency) + "\n\x1b\x61\x01";
                receipt += "Terima Kasih!\n\n\n\x1d\x56\x42\x00";

                // Kirim ke Bridge lokal di PC Kasir
                fetch('http://127.0.0.1:18212/api/print/raw', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ data: receipt, doc_name: frm.doc.name })
                })
                .then(r => r.json())
                .then(res => {
                    if (res.status === 'success') frappe.show_alert({ message: __('Struk berhasil dicetak!'), indicator: 'green' });
                    else frappe.msgprint(__('Gagal mencetak: ' + res.detail));
                })
                .catch(() => frappe.msgprint(__('Hardware Bridge tidak aktif di http://127.0.0.1:18212')));
            }, __('Aksi Cepat'));

            // Tombol Buka Laci Kasir
            frm.add_custom_button(__('💵 Buka Laci Kasir'), function() {
                fetch('http://127.0.0.1:18212/api/cashdrawer/open', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ pin: 2 })
                }).then(() => frappe.show_alert({ message: __('Laci kasir terbuka!'), indicator: 'green' }));
            }, __('Aksi Cepat'));
        }

        // Tombol Baca Timbangan
        frm.add_custom_button(__('⚖️ Timbang Item Aktif'), function() {
            frappe.show_alert({ message: __('Menimbang... Harap tunggu stabil'), indicator: 'blue' });
            fetch('http://127.0.0.1:18212/api/scale/stable-read?timeout=4.0', { method: 'POST' })
                .then(r => r.json())
                .then(res => {
                    if (res.status === 'success') {
                        frappe.msgprint({
                            title: __('Hasil Timbangan'),
                            indicator: 'green',
                            message: `Berat: ${res.data.weight} ${res.data.unit} (Stabil)`
                        });
                    } else {
                        frappe.msgprint(__('Timbangan belum stabil: ' + res.message));
                    }
                });
        }, __('Aksi Cepat'));
    }
});

3. Hook POS Checkout Otomatis (POS Awesome & ERPNext POS)

Script ini mendengarkan event submit POS invoice untuk langsung mencetak struk dan menendang laci uang saat kasir selesai memproses pembayaran: 

$(document).on('pos_invoice_submitted', function(event, invoice_doc) {
    if (!invoice_doc) return;

    // 1. Buka laci kasir otomatis
    fetch('http://127.0.0.1:18212/api/cashdrawer/open', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ pin: 2 })
    }).catch(() => {});

    // 2. Format Struk Thermal ESC/POS
    let receipt = "\x1b\x40\x1b\x61\x01\x1d\x21\x11" + (invoice_doc.company || "STRUK KASIR") + "\n\x1d\x21\x00";
    receipt += "----------------------------------------\n\x1b\x61\x00";
    receipt += "No: " + invoice_doc.name + "\n";
    receipt += "Tgl: " + invoice_doc.posting_date + "\nKasir: " + (invoice_doc.owner || "") + "\n----------------------------------------\n";

    (invoice_doc.items || []).forEach(item => {
        receipt += `${item.qty}x ${item.item_name} - ${Number(item.amount).toLocaleString('id-ID')}\n`;
    });

    receipt += "----------------------------------------\n\x1b\x61\x02";
    receipt += "TOTAL: Rp " + Number(invoice_doc.grand_total).toLocaleString('id-ID') + "\n";
    receipt += "----------------------------------------\n\x1b\x61\x01";
    receipt += "Terima Kasih!\n\n\n\x1d\x56\x42\x00";

    // 3. Kirim ke Bridge lokal di PC Kasir
    fetch('http://127.0.0.1:18212/api/print/raw', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ data: receipt, doc_name: invoice_doc.name })
    }).catch(err => console.warn('Gagal print POS:', err));
});

4. Penggunaan via REST API (cURL / Backend Python / PHP / C#)

Anda juga dapat memanggil bridge dari script command line, backend lokal, atau desktop app: 

# 1. Ambil daftar printer
curl -X GET http://127.0.0.1:18212/api/printers

# 2. Cetak RAW teks / ESC/POS
curl -X POST http://127.0.0.1:18212/api/print/raw \
  -H "Content-Type: application/json" \
  -d '{"printer": "Godex G500 GZPL", "data": "Halo Dunia Kasir!\n\n\n"}'

# 3. Cetak PDF Silent
curl -X POST http://127.0.0.1:18212/api/print/pdf \
  -H "Content-Type: application/json" \
  -d '{"pdf_data": "http://127.0.0.1:8088/api/invoice/INV-001.pdf"}'

# 4. Ambil Berat Timbangan
curl -X GET http://127.0.0.1:18212/api/weight

# 5. Buka Laci Kasir
curl -X POST http://127.0.0.1:18212/api/cashdrawer/open -H "Content-Type: application/json" -d '{"pin": 2}'
